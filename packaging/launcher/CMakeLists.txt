# The Linux launcher / setup window: "GoldenEye 007" in the release root. It
# starts the game from resources/runtime, importing the player's own Xbox 360
# package first when needed (an SDL3 + Dear ImGui window, or the terminal with
# --cli / without a display). SDL3 and ImGui come from the SDK tree; nothing
# else of the SDK is used. (The Windows launcher is built by
# packaging/Build-Launcher.ps1 from the same core.cpp/stfs.cpp.)

# ge.toml and player-names.txt are embedded as byte arrays, the way the Windows
# launcher carries them as resources (ge_embedded.h, generated at configure time).
set(_ge_pack_dir "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(_ge_embed_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")

function(_ge_embed_bytes out_var name file)
    file(READ "${file}" _hex HEX)
    string(LENGTH "${_hex}" _hex_length)
    math(EXPR _size "${_hex_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
    # A trailing NUL keeps the array non-empty and makes it a C string too (not counted in _size).
    set(${out_var}
        "static const unsigned char ge_embedded_${name}[] = {${_bytes}0x00};\nstatic const std::size_t ge_embedded_${name}_size = ${_size};\n"
        PARENT_SCOPE)
endfunction()

_ge_embed_bytes(_ge_embed_config default_config "${_ge_pack_dir}/ge.toml")
_ge_embed_bytes(_ge_embed_names player_names "${_ge_pack_dir}/player-names.txt")
# player-names.txt is already a configure dependency of the top-level
# CMakeLists.txt (listing a file twice makes Ninja reject build.ninja).
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_ge_pack_dir}/ge.toml")
file(GENERATE OUTPUT "${_ge_embed_dir}/ge_embedded.h"
    CONTENT "// Generated by packaging/launcher/CMakeLists.txt from packaging/ge.toml and packaging/player-names.txt.\n#pragma once\n#include <cstddef>\n${_ge_embed_config}${_ge_embed_names}")

set(_ge_imgui_backends "${CMAKE_SOURCE_DIR}/rexglue-sdk/thirdparty/imgui/backends")
# The vendored ImGui backends are not part of any SDK target; compile them here, quietly.
set_source_files_properties(
    "${_ge_imgui_backends}/imgui_impl_sdl3.cpp"
    "${_ge_imgui_backends}/imgui_impl_sdlrenderer3.cpp"
    PROPERTIES COMPILE_OPTIONS "-w")
add_executable(ge_launcher core.cpp stfs.cpp main_cli.cpp setup_window_sdl.cpp
    "${_ge_imgui_backends}/imgui_impl_sdl3.cpp"
    "${_ge_imgui_backends}/imgui_impl_sdlrenderer3.cpp")
set_target_properties(ge_launcher PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
    OUTPUT_NAME "GoldenEye 007"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/launcher")
target_include_directories(ge_launcher PRIVATE "${_ge_embed_dir}")
target_compile_definitions(ge_launcher PRIVATE "GE_VERSION=\"${GE_VERSION}\"")
target_include_directories(ge_launcher PRIVATE
    "${_ge_imgui_backends}"
    "${CMAKE_SOURCE_DIR}/rexglue-sdk/thirdparty/stb")
target_link_libraries(ge_launcher PRIVATE imgui::imgui SDL3::SDL3-static)
target_compile_options(ge_launcher PRIVATE -Wall -Wextra)
